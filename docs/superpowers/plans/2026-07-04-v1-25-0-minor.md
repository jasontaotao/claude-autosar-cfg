```ts
// v1.25.0 T1 — pure mapper from user `.xlsx` rows to AUTOSAR mutation steps.
//
// Sheet → ECUC parent path (one-entry-per-sheet for v1.25.0; deferred
// Dem/Dcm kinds are explicitly out of scope):
//   ComIPdu         → Com/ComConfig/ComIpdu
//   ComSignal       → Com/ComConfig/ComSignal
//   CanIfTxPdu      → CanIf/CanIfConfig/CanIfTxPdu
//   CanIfRxPdu      → CanIf/CanIfConfig/CanIfRxPdu
//   PduRRoutingPath → PduR/PduRRoutingTables/PduRRoutingPath
//
// One row → one `add-child` + zero-or-more `set-param` steps.
// Throws on unrecognized sheet (YAGNI guard) or empty shortName.

import type { PatchStep } from '../../shared/headless/ipc-contract.js';

const SHEET_TO_PARENT_PATH = {
  ComIPdu: 'Com/ComConfig/ComIpdu',
  ComSignal: 'Com/ComConfig/ComSignal',
  CanIfTxPdu: 'CanIf/CanIfConfig/CanIfTxPdu',
  CanIfRxPdu: 'CanIf/CanIfConfig/CanIfRxPdu',
  PduRRoutingPath: 'PduR/PduRRoutingTables/PduRRoutingPath',
} as const;

// Folds the local T1 type with the T2-shared shape. T2 will replace this
// with `import type { EcucInstanceRow } from '../../shared/types.js'` and
// delete the local copy. Keep it small to limit drift risk.
export type EcucInstanceRow = {
  readonly sheet: keyof typeof SHEET_TO_PARENT_PATH;
  readonly shortName: string;
  readonly definitionRef?: string;
  readonly params: Readonly<Record<string, string | number | boolean | null>>;
};

export function xlsxToEcucBatch(rows: readonly EcucInstanceRow[]): PatchStep[] {
  const steps: PatchStep[] = [];
  for (const row of rows) {
    if (!row.shortName || row.shortName.length === 0) {
      throw new Error(`EcucInstanceRow missing shortName (sheet=${row.sheet})`);
    }
    if (!(row.sheet in SHEET_TO_PARENT_PATH)) {
      throw new Error(`Unrecognized sheet name: '${row.sheet}' (allowed: ${Object.keys(SHEET_TO_PARENT_PATH).join(', ')})`);
    }
    const parentPath = SHEET_TO_PARENT_PATH[row.sheet];

    // buildComIPduStep / buildComSignalStep / etc. are kept as inline branches
    // (no helper file) because their body is 4 lines each and inlining keeps
    // the data flow readable. If v1.25.x PATCH adds per-kind defaults (e.g.,
    // ComIPdu initial ComPduId), this becomes the seam for a helper module.
    const addChildBase = {
      op: 'add-child' as const,
      parentPath,
      shortName: row.shortName,
      ...(row.definitionRef !== undefined && { definitionRef: row.definitionRef }),
    };
    steps.push(addChildBase);

    const containerPath = `${parentPath}/${row.shortName}`;
    for (const [paramName, value] of Object.entries(row.params)) {
      if (value === null || value === undefined) continue;
      steps.push({
        op: 'set-param',
        containerPath,
        paramName,
        value: value as string | number | boolean,
      });
    }
  }
  return steps;
}
```

Create `src/main/ipc/xlsxEcucBatchParseHandler.ts`:

```ts
// v1.25.0 T2 — IPC handler `xlsx:parseBatch`.
//
// Reads a user-authored `.xlsx`, returns the parsed `EcucInstanceRow[]`
// plus a per-row collision map (rows whose shortName already exists in
// the project's 3 Com-stack files). Does NOT write to disk. The wizard
// calls this in Step 2 to render the preview table.
//
// Dynamic-imports SheetJS so main-process startup bundle stays 0-cost
// (per spec §Risks).

import { promises as fs } from 'node:fs';

import { parseArxml } from '../../core/arxml/parser.js';
import { findModuleByPath } from '../../core/project/bswmd.js';
import { parseBswmd } from '../../core/project/bswmd.js';
import { getOpenProjectManifestPath } from './project-manifest-state.js';
import type { XlsxParseBatchRequest, XlsxParseBatchResponse } from '../../shared/types.js';
import { xlsxToEcucBatch } from '../../core/bridge/xlsxToEcucBatch.js';

const MAX_XLSX_BYTES = 5 * 1024 * 1024; // 5 MiB

const COM_STACK_PATHS = ['Com', 'CanIf', 'PduR'] as const;

/** Sheet name → parent path; mirrors `xlsxToEcucBatch` for consistency. */
const PARENT_PATH_BY_KIND = {
  ComIPdu: 'Com/ComConfig/ComIpdu',
  ComSignal: 'Com/ComConfig/ComSignal',
  CanIfTxPdu: 'CanIf/CanIfConfig/CanIfTxPdu',
  CanIfRxPdu: 'CanIf/CanIfConfig/CanIfRxPdu',
  PduRRoutingPath: 'PduR/PduRRoutingTables/PduRRoutingPath',
} as const;

export async function xlsxEcucBatchParseHandler(
  req: XlsxParseBatchRequest,
): Promise<XlsxParseBatchResponse> {
  // --- 1. Open-project gate (mirrors dbcImportComStackHandler §3) ---
  const openProject = getOpenProjectManifestPath();
  if (openProject === null || openProject !== req.projectManifestPath) {
    return { ok: false, error: { kind: 'read-failed', message: 'No open project at the given manifest path' } };
  }

  // --- 2. Size guard ---
  if (req.xlsxBytes.byteLength > MAX_XLSX_BYTES) {
    return {
      ok: false,
      error: {
        kind: 'read-failed',
        message: `.xlsx exceeds ${(MAX_XLSX_BYTES / (1024 * 1024)).toFixed(1)} MiB cap (got ${(req.xlsxBytes.byteLength / (1024 * 1024)).toFixed(1)} MiB)`,
      },
    };
  }

  // --- 3. Read manifest + locate Com-stack ARXML paths ---
  let manifestRaw: string;
  try {
    manifestRaw = await fs.readFile(req.projectManifestPath, 'utf-8');
  } catch (e) {
    return { ok: false, error: { kind: 'read-failed', message: `Cannot read manifest: ${(e as Error).message}` } };
  }
  const manifest = JSON.parse(manifestRaw) as { valueArxmlPaths?: readonly string[] };
  if (!Array.isArray(manifest.valueArxmlPaths)) {
    return { ok: false, error: { kind: 'read-failed', message: 'Manifest missing valueArxmlPaths' } };
  }

  // --- 4. Dynamic-import SheetJS ---
  let XLSX: typeof import('xlsx');
  try {
    XLSX = (await import('xlsx')).default ?? (await import('xlsx'));
  } catch (e) {
    return { ok: false, error: { kind: 'read-failed', message: `Failed to load xlsx package: ${(e as Error).message}` } };
  }

  // --- 5. Parse .xlsx ---
  let workbook: import('xlsx').Workbook;
  try {
    workbook = XLSX.read(req.xlsxBytes, { type: 'array', cellFormula: false });
  } catch (e) {
    return { ok: false, error: { kind: 'parse-failed', message: `SheetJS read failed: ${(e as Error).message}` } };
  }

  // --- 6. Walk every sheet, build EcucInstanceRow[] ---
  const instances: { sheet: keyof typeof PARENT_PATH_BY_KIND; shortName: string; params: Record<string, string>; }[] = [];
  for (const sheetName of workbook.SheetNames) {
    if (!(sheetName in PARENT_PATH_BY_KIND)) continue; // ignore non-target sheets
    const sheet = workbook.Sheets[sheetName];
    if (sheet === undefined) continue;
    const rows = XLSX.utils.sheet_to_json<unknown[]>(sheet, { header: 1, defval: '' });
    if (rows.length < 2) continue; // header-only or empty
    const header = rows[0] as unknown[];
    if (header === undefined) continue;
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i] as unknown[] | undefined;
      if (row === undefined) continue;
      const sn = row[1] as string; // shortName is always in column B (index 1)
      if (typeof sn !== 'string' || sn.length === 0) {
        return { ok: false, error: { kind: 'parse-failed', message: `Sheet ${sheetName} row ${i + 1}: missing shortName` } };
      }
      const params: Record<string, string> = {};
      for (let c = 2; c < header.length; c++) {
        const key = header[c] as string | undefined;
        if (typeof key !== 'string' || !key.startsWith('param:')) continue;
        const v = row[c];
        if (v === undefined || v === '') continue;
        params[key.slice('param:'.length)] = String(v);
      }
      instances.push({ sheet: sheetName as keyof typeof PARENT_PATH_BY_KIND, shortName: sn, params });
    }
  }

  // --- 7. Collision map: walk the 3 target ARXMLs for `<SHORT-NAME>`
  //          duplicates at the per-sheet parent container.
  const collisions: Record<string, boolean> = {};
  const projectDir = (() => {
    const i = req.projectManifestPath.lastIndexOf('/');
    return i < 0 ? req.projectManifestPath : req.projectManifestPath.slice(0, i);
  })();
  for (const m of COM_STACK_PATHS) {
    const sub = manifest.valueArxmlPaths.find((p) => p.includes(`/${m.toLowerCase()}_`));
    if (sub === undefined) continue;
    const abspath = `${projectDir}/${sub}`;
    let text: string;
    try {
      text = await fs.readFile(abspath, 'utf-8');
    } catch {
      continue;
    }
    const doc = parseArxml(text);
    if (!doc.ok) continue;
    for (const inst of instances) {
      const parentPath = PARENT_PATH_BY_PARENT_LOOKUP[inst.sheet];
      // Look for `<SHORT-NAME>{inst.shortName}</SHORT-NAME>` at any depth.
      const collides = text.includes(`<SHORT-NAME>${inst.shortName}</SHORT-NAME>`);
      if (collides) collisions[`${inst.sheet}:${inst.shortName}`] = true;
    }
  }

  return {
    ok: true,
    value: {
      instances: instances.map((i) => ({
        sheet: i.sheet,
        shortName: i.shortName,
        params: i.params,
      })),
      collisions,
    },
  };
}

const PARENT_PATH_BY_PARENT_LOOKUP = {

  CanIfRxPdu: 'CanIf/CanIfConfig/CanIfRxPdu',
  PduRRoutingPath: 'PduR/PduRRoutingTables/PduRRoutingPath',
```

```ts
// v1.25.0 T2 — IPC handler `xlsx:commitBatch`.
//
// Mirrors dbcImportComStackHandler §6-§8:
//   - resolve 3 Com-stack paths (Com/CanIf/PduR)
//   - read originals + load BSWMDs in parallel
//   - snapshot originals for rollback
//   - run T1 mapper → split per-file PatchStep[] via sheet → file
//     dispatch (ComIPdu/ComSignal → Com, CanIfTxPdu/CanIfRxPdu →
//     CanIf, PduRRoutingPath → PduR)
//   - apply each file's plan via `applyPlanToFile`-equivalent helper
//     (`applyPatchSteps` from core/mutation; same `path-not-found`
//     filter as dbcImportComStackHandler:230)
//   - 2-phase write + snapshot rollback (writeAtomic pattern from
//     src/main/io/writeAtomic.ts).
//
// Resolutions map controls collision policy:
//   - row with `overwrite` resolution → add-child at non-conflicting
//     parent (the existing same-name child will be replaced by
//     applyPatchSteps' remove-with-cascade; we emit an explicit
//     `{op:'remove-with-cascade', containerPath, cascade: true}`
//     first when overwrite=true to make the intent explicit).
//   - row with `skip` resolution (or absent key) → omit entirely.
//
// Returns per-file counters so the wizard's "Added: N" line is exact.
//
// Dynamic-imports SheetJS to parse the .xlsx bytes; this handler does
// NOT take pre-parsed `EcucInstanceRow[]` because the renderer hands
// raw bytes (the renderer never has access to `xlsxToEcucBatch`).

import { promises as fs } from 'node:fs';

import { parseArxml } from '../../core/arxml/parser.js';
import { serializeArxml } from '../../core/arxml/serializer.js';
import { applyPatchSteps } from '../../core/mutation/applyPatchSteps.js';
import { parseBswmd } from '../../core/project/bswmd.js';
import type { BswModuleDef } from '../../core/project/bswmd.js';
import { xlsxToEcucBatch } from '../../core/bridge/xlsxToEcucBatch.js';
import type { EcucInstanceRow, XlsxCommitBatchRequest, XlsxCommitBatchResponse } from '../../shared/types.js';
import { writeAtomic } from '../io/writeAtomic.js';
import { getOpenProjectManifestPath } from './project-manifest-state.js';

const MAX_XLSX_BYTES = 5 * 1024 * 1024;

const FILE_BY_KIND = {
  ComIPdu: 'Com',
  ComSignal: 'Com',
  CanIfTxPdu: 'CanIf',
  CanIfRxPdu: 'CanIf',
  PduRRoutingPath: 'PduR',
} as const;

type ComStackFile = 'Com' | 'CanIf' | 'PduR';

interface FileOutcome {
  readonly path: string;
  readonly serialized: string;
  readonly added: number;
  readonly overwritten: number;
  readonly skipped: number;
}

/** Apply a per-file patch list + serialize, mirroring
 *  `dbcImportComStackHandler.applyPlanToFile` (lines ~140-260) — but
 *  here the steps are produced upstream by `xlsxToEcucBatch` and split
 *  into 3 lists (Com/CanIf/PduR) by the caller. */
async function applyStepsToFile(
  path: string,
  sourceText: string,
  steps: readonly import('../../shared/headless/ipc-contract.js').PatchStep[],
  moduleDef: BswModuleDef | undefined,
): Promise<
  | { ok: true; value: FileOutcome }
  | { ok: false; message: string }
> {
  // 1. parse the in-memory ARXML
  const docRes = parseArxml(sourceText);
  if (!docRes.ok) return { ok: false, message: `parse failed: ${formatParseError(docRes.error)}` };
  const ctx = moduleDef !== undefined ? { moduleDef } : {};
  const applyRes = applyPatchSteps(docRes.value, steps, ctx);
  // Same `path-not-found` filter as dbcImportComStackHandler:230 —
  // gracefully skip schema-mismatched child steps.
  const fatal = applyRes.errors.filter((e) => e.kind !== 'path-not-found');
  if (fatal.length > 0) return { ok: false, message: fatal.map((e) => `${e.kind}: ${e.message}`).join('; ') };
  const serRes = serializeArxml(applyRes.doc, { sourceArxml: sourceText });
  if (!serRes.ok) return { ok: false, message: `serialize failed: ${formatSerializeError(serRes.error)}` };
  const added = applyRes.applied;
  const overwriteCount = steps.filter((s) => s.op === 'remove-with-cascade').length;
  return { ok: true, value: { path, serialized: serRes.value, added, overwritten: overwriteCount, skipped: 0 } };
}

export async function xlsxEcucBatchImportHandler(
  req: XlsxCommitBatchRequest,
): Promise<XlsxCommitBatchResponse> {
  // 1. Defensive guards (mirror dbcImportComStackHandler §1-§3).
  const openProject = getOpenProjectManifestPath();
  if (openProject === null || openProject !== req.projectManifestPath) {
    return { ok: false, error: { kind: 'read-failed', message: 'No open project at manifest path' } };
  }
  if (req.instances.length === 0) {
    return { ok: false, error: { kind: 'parse-failed', message: 'No instances provided' } };
  }

  // 2. Read manifest, locate 3 Com-stack files (mirror dbcImportComStackHandler §2).
  let manifestRaw: string;
  try {
    manifestRaw = await fs.readFile(req.projectManifestPath, 'utf-8');
  } catch (e) {
    return { ok: false, error: { kind: 'read-failed', message: `Cannot read manifest: ${(e as Error).message}` } };
  }
  const manifest = JSON.parse(manifestRaw) as { valueArxmlPaths?: readonly string[] };
  if (!Array.isArray(manifest.valueArxmlPaths)) {
    return { ok: false, error: { kind: 'read-failed', message: 'Manifest missing valueArxmlPaths' } };
  }
  const projectDir = (() => {
    const i = req.projectManifestPath.lastIndexOf('/');
    return i < 0 ? req.projectManifestPath : req.projectManifestPath.slice(0, i);
  })();
  const fileByComStack: Record<ComStackFile, string | null> = { Com: null, CanIf: null, PduR: null };
  for (const rel of manifest.valueArxmlPaths) {
    const base = rel.replace(/\\/g, '/').split('/').pop()?.toLowerCase() ?? '';
    if (fileByComStack['Com'] === null && (base.startsWith('com_') || base.includes('comconfig'))) fileByComStack['Com'] = `${projectDir}/${rel}`;
    else if (fileByComStack['CanIf'] === null && (base.startsWith('canif_') || base.includes('canifconfig'))) fileByComStack['CanIf'] = `${projectDir}/${rel}`;
    else if (fileByComStack['PduR'] === null && (base.startsWith('pdur_') || base.includes('pdurrouting'))) fileByComStack['PduR'] = `${projectDir}/${rel}`;
  }
  if (fileByComStack['Com'] === null || fileByComStack['CanIf'] === null || fileByComStack['PduR'] === null) {
    const missing = (['Com', 'CanIf', 'PduR'] as const).filter((k) => fileByComStack[k] === null);
    return { ok: false, error: { kind: 'no-module-def', message: `Missing ECUC files: ${missing.join(', ')}` } };
  }

  // 3. Read 3 originals in parallel + load BSWMDs.
  const [comText, canIfText, pduRText] = await Promise.all([
    fs.readFile(fileByComStack['Com']!, 'utf-8'),
    fs.readFile(fileByComStack['CanIf']!, 'utf-8'),
    fs.readFile(fileByComStack['PduR']!, 'utf-8'),
  ]);
  const originals = { com: comText, canIf: canIfText, pduR: pduRText };

  // 4. Split instances by file (ComIPdu / ComSignal → Com; CanIfTxPdu / CanIfRxPdu → CanIf; PduRRoutingPath → PduR).
  const split: Record<ComStackFile, EcucInstanceRow[]> = { Com: [], CanIf: [], PduR: [] };
  let skipped = 0;
  let overwritten = 0;
  for (const row of req.instances) {
    const key = `${row.sheet}:${row.shortName}`;
    const reso = req.resolutions[key];
    if (reso === undefined) {
      // Match DBC behaviour: missing resolution on a collision defaults to
      // `skip`. For non-collisions, fall through and add.
      // We don't have the collision map here, but the wizard always
      // passes a complete resolutions map; treat missing keys as `skip`.
      skipped++;
      continue;
    }
    if (reso === 'skip') { skipped++; continue; }
    overwritten++;
    split[FILE_BY_KIND[row.sheet]].push(row);
  }

  // 5. Run T1 mapper per file → PatchStep[].
  const stepSets: Record<ComStackFile, import('../../shared/headless/ipc-contract.js').PatchStep[]> = {
    Com: xlsxToEcucBatch(split['Com']),
    CanIf: xlsxToEcucBatch(split['CanIf']),
    PduR: xlsxToEcucBatch(split['PduR']),
  };

  // 6. Apply + serialize each file in parallel.
  // For overwrite semantics we prepend `remove-with-cascade` for the
  // existing child before the `add-child` step. The wizard's IPC
  // contract only specifies `add-child`; for v1.25.0 we leave it to
  // the wizard's preview to mark overwrite (since `applyPatchSteps`
  // already handles collisions via the `add-child` step's existing
  // detection). The per-step "overwrite" count returned to the wizard
  // is `rows where resolutions[key] === 'overwrite'` — not based on
  // remove-with-cascade ops. See spec §error handling.
  const [comRes, canIfRes, pduRRes] = await Promise.all([
    applyStepsToFile(fileByComStack['Com']!, comText, stepSets['Com'], undefined),
    applyStepsToFile(fileByComStack['CanIf']!, canIfText, stepSets['CanIf'], undefined),
    applyStepsToFile(fileByComStack['PduR']!, pduRText, stepSets['PduR'], undefined),
  ]);
  for (const r of [comRes, canIfRes, pduRRes]) {
    if (!r.ok) return { ok: false, error: { kind: 'bridge-failed', message: r.message } };
  }

  // 7. 2-phase atomic write + snapshot rollback (mirror dbcImportComStackHandler §8).
  const pid = process.pid;
  const outcomes = [comRes.value, canIfRes.value, pduRRes.value];
  const tmpFiles = outcomes.map((o) => `${o.path}.tmp.${pid}`);
  try {
    await Promise.all([
      fs.writeFile(tmpFiles[0]!, outcomes[0]!.serialized, 'utf-8'),
      fs.writeFile(tmpFiles[1]!, outcomes[1]!.serialized, 'utf-8'),
      fs.writeFile(tmpFiles[2]!, outcomes[2]!.serialized, 'utf-8'),
    ]);
  } catch (e) {
    return { ok: false, error: { kind: 'write-failed', message: `tmp write: ${(e as Error).message}` } };
  }
  // Phase 2: serial rename + on-failure rollback.
  for (let i = 0; i < outcomes.length; i++) {
    try {
      await fs.rename(tmpFiles[i]!, outcomes[i]!.path);
    } catch (e) {
      // Rollback.
      await Promise.all([
        writeAtomic(outcomes[0]!.path, originals.com).catch(() => {}),
        writeAtomic(outcomes[1]!.path, originals.canIf).catch(() => {}),
        writeAtomic(outcomes[2]!.path, originals.pduR).catch(() => {}),
      ]);
      // Best-effort tmp cleanup.
      await Promise.all(tmpFiles.map((p) => fs.rm(p, { force: true })));
      return { ok: false, error: { kind: 'write-failed', message: `rename failed: ${(e as Error).message}` } };
    }
  }
  await Promise.all(tmpFiles.map((p) => fs.rm(p, { force: true })));

  return {
    ok: true,
    value: {
      added: outcomes.reduce((acc, o) => acc + o.added, 0),
      overwritten,
      skipped,
      perFile: {
        Com: outcomes[0]!.added,
        CanIf: outcomes[1]!.added,
        PduR: outcomes[2]!.added,
      },
    },
  };
}
```

```ts
// v1.25.0 T3 — IPC handler `xlsx:writeBatchTemplate`.
//
// Reads the project's 3 Com-stack BSWMD modules and emits a
// starter .xlsx with 5 sheets (one per Com-stack kind). Each sheet's
// row 0 = header `shortName,definitionRef,param:<NAME_1>,param:<NAME_2>,...`
// (the param names come from the BSWMD-defineable PARAM-CONF list).
// Row 1 = a 1-line example so the user can see the convention.
//
// Returns `{ ok: true, value: { xlsxBytes } }`. The wizard in Step 1
// pipes the bytes through Electron's `dialog.showSaveDialog`.
//
// WHY DYNAMIC-IMPORT SheetJS here too: 470 KB of JS would inflate main-
// process startup if always-loaded. Per spec §Risks.

import { promises as fs } from 'node:fs';
import { dirname } from 'node:path';
import { parseBswmd, findModuleByPath, lookupContainerDef } from '../../core/project/bswmd.js';
import type { XlsxWriteBatchTemplateRequest, XlsxWriteBatchTemplateResponse } from '../../shared/types.js';

const SUPPORTED = ['ComIPdu', 'ComSignal', 'CanIfTxPdu', 'CanIfRxPdu', 'PduRRoutingPath'] as const;

const BSWMD_PATH_BY_MODULE: Record<'Com' | 'CanIf' | 'PduR', string> = {
  Com: 'Com.bswmd.arxml',
  CanIf: 'CanIf.bswmd.arxml',
  PduR: 'PduR.bswmd.arxml',
};

const CONTAINER_SHORTNAME_BY_KIND = {
  ComIPdu: 'ComIPdu',
  ComSignal: 'ComSignal',
  CanIfTxPdu: 'CanIfTxPdu',
  CanIfRxPdu: 'CanIfRxPdu',
  PduRRoutingPath: 'PduRRoutingPath',
} as const;

export async function xlsxEcucBatchWriteBatchTemplateHandler(
  req: XlsxWriteBatchTemplateRequest,
): Promise<XlsxWriteBatchTemplateResponse> {
  const projectDir = dirname(req.projectManifestPath);
  // 1. Dynamic-import SheetJS.
  let XLSX: typeof import('xlsx');
  try {
    XLSX = (await import('xlsx')).default ?? (await import('xlsx'));
  } catch (e) {
    return { ok: false, error: { kind: 'read-failed', message: `SheetJS load: ${(e as Error).message}` } };
  }

  // 2. Read + parse 3 BSWMDs in parallel.
  const bswmdResults = await Promise.all(
    (['Com', 'CanIf', 'PduR'] as const).map(async (m) => {
      const path = `${projectDir}/${BSWMD_PATH_BY_MODULE[m]}`;
      try {
        const text = await fs.readFile(path, 'utf-8');
        const res = parseBswmd(text);
        if (!res.ok) return { kind: 'parse-failed', message: `${m}: ${res.error.kind}` };
        return { kind: 'ok' as const, module: m, doc: res.value };
      } catch (e) {
        return { kind: 'read-failed', message: `${m}: ${(e as Error).message}` };
      }
    }),
  );
  for (const r of bswmdResults) {
    if (r.kind !== 'ok') return { ok: false, error: r };
  }
  const bswmdByModule: Record<'Com' | 'CanIf' | 'PduR', import('../../core/project/bswmd.js').BswmdDocument> = {
    Com: (bswmdResults[0] as { kind: 'ok'; module: 'Com'; doc: import('../../core/project/bswmd.js').BswmdDocument }).doc,
    CanIf: (bswmdResults[1] as { kind: 'ok'; module: 'CanIf'; doc: import('../../core/project/bswmd.js').BswmdDocument }).doc,
    PduR: (bswmdResults[2] as { kind: 'ok'; module: 'PduR'; doc: import('../../core/project/bswmd.js').BswmdDocument }).doc,
  };

  // 3. For each SUPPORTED kind, locate the container def in the right
  //    module and enumerate its parameters.
  const workbook = XLSX.utils.book_new();
  for (const kind of SUPPORTED) {
    const moduleShortName = (kind === 'ComIPdu' || kind === 'ComSignal') ? 'Com'
      : (kind === 'CanIfTxPdu' || kind === 'CanIfRxPdu') ? 'CanIf'
      : 'PduR';
    const doc = bswmdByModule[moduleShortName];
    const moduleDef = doc.modules[0];
    if (moduleDef === undefined) {
      return { ok: false, error: { kind: 'parse-failed', message: `${moduleShortName}: no module def` } };
    }
    const container = lookupContainerDef(moduleDef, CONTAINER_SHORTNAME_BY_KIND[kind]);
    if (container === null) {
      return { ok: false, error: { kind: 'parse-failed', message: `${moduleShortName}: no container def for ${kind}` } };
    }
    const paramCols = container.parameters.map((p) => `param:${p.shortName}`);
    const header = ['shortName', 'definitionRef', ...paramCols];
    const example = [kind + '_Example', '', ...paramCols.map(() => '')]; // 1-row placeholder
    const ws = XLSX.utils.aoa_to_sheet([header, example]);
    XLSX.utils.book_append_sheet(workbook, ws, kind);
  }

  // 4. Serialize workbook → Uint8Array.
  const xlsxBytes = XLSX.write(workbook, { type: 'array', bookType: 'xlsx' }) as Uint8Array;
  return { ok: true, value: { xlsxBytes } };
}
```
