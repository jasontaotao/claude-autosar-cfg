```ts
// PHASE 0: Snapshot originals (in-memory, <100KB for demo-ecu)
const originals = { com: comContent, canIf: canIfContent, pduR: pduRContent };

// ... (existing parse-apply-serialize for each file, populating newCom/newCanIf/newPduR)

// PHASE 1: Write 3 tmp files in parallel
const pid = process.pid;
const tmpCom = `${comPath}.tmp.${pid}`;
const tmpCanIf = `${canIfPath}.tmp.${pid}`;
const tmpPduR = `${pduRPath}.tmp.${pid}`;

try {
  await Promise.all([
    fs.writeFile(tmpCom, newCom, 'utf-8'),
    fs.writeFile(tmpCanIf, newCanIf, 'utf-8'),
    fs.writeFile(tmpPduR, newPduR, 'utf-8'),
  ]);
} catch (e) {
  // Tmp write failed — clean up any partial tmp files
  await Promise.allSettled([
    fs.unlink(tmpCom).catch(() => {}),
    fs.unlink(tmpCanIf).catch(() => {}),
    fs.unlink(tmpPduR).catch(() => {}),
  ]);
  return {
    ok: false,
    error: {
      kind: 'write-failed',
      message: e instanceof Error ? e.message : String(e),
      rolledBack: true,
    },
  };
}

// PHASE 2: Atomic rename in serial (so we can rollback between failures).
// Step 6 refactors this to per-rename try-catch; the snippet below is the
// Step 5 high-level version that the implementer will replace.
const renames: ReadonlyArray<{ tmp: string; target: string; file: 'com' | 'canIf' | 'pduR' }> = [
  { tmp: tmpCom, target: comPath, file: 'com' },
  { tmp: tmpCanIf, target: canIfPath, file: 'canIf' },
  { tmp: tmpPduR, target: pduRPath, file: 'pduR' },
];
let phase2Failed: { file: 'com' | 'canIf' | 'pduR'; message: string } | null = null;
for (const r of renames) {
  try {
    await fs.rename(r.tmp, r.target);
  } catch (e) {
    phase2Failed = { file: r.file, message: e instanceof Error ? e.message : String(e) };
    break;
  }
}
```

```ts
// PHASE 3: Best-effort rollback using snapshot
if (phase2Failed !== null) {
  const rollbackResults = await Promise.allSettled([
    writeAtomic(comPath, originals.com),
    writeAtomic(canIfPath, originals.canIf),
    writeAtomic(pduRPath, originals.pduR),
  ]);
  // Clean up any leftover tmp files
  await Promise.allSettled([
    fs.unlink(tmpCom).catch(() => {}),
    fs.unlink(tmpCanIf).catch(() => {}),
    fs.unlink(tmpPduR).catch(() => {}),
  ]);
  // rolledBack = true only if ALL 3 rollbacks succeeded
  const rolledBack = rollbackResults.every((r) => r.status === 'fulfilled');
  return {
    ok: false,
    error: { kind: 'write-failed', message: phase2Failed.message, rolledBack },
  };
}

// Success: clean up tmp files (should already be moved, but defensive)
await Promise.allSettled([
  fs.unlink(tmpCom).catch(() => {}),
  fs.unlink(tmpCanIf).catch(() => {}),
  fs.unlink(tmpPduR).catch(() => {}),
]);

return {
  ok: true,
  value: { addedCounts: { com: comAdds, canIf: canIfAdds, pduR: pduRAdds } },
```

```ts
// ... etc for all 7 Chinese locale tables

import type { Messages } from './types.js';
import type { Locale } from './types.js';

export type MessagesEn = Messages;
export type MessagesZhCN = Messages;

export const DEFAULT_LOCALE: Locale = 'zh-CN';
export const SUPPORTED_LOCALES: readonly Locale[] = ['zh-CN', 'en'] as const;

export const MESSAGES_BY_LOCALE: Readonly<Record<Locale, Messages>> = {
  'zh-CN': {
    ...AppZhCN,
    ...DialogZhCN,
    ...EditorZhCN,
    ...ValidationZhCN,
    ...DbcZhCN,
    ...OdxZhCN,
    ...MiscZhCN,
  },
  en: {
    ...AppEn,
    ...DialogEn,
    ...EditorEn,
    ...ValidationEn,
    ...DbcEn,
    ...OdxEn,
    ...MiscEn,
  },
} as const;

export function t(
  locale: Locale,
  key: keyof Messages,
  params?: Record<string, string | number>,
): string {
  const template = MESSAGES_BY_LOCALE[locale][key];
  if (params === undefined) return template;
  return template.replace(/\{(\w+)\}/g, (_match, name) => {
    const value = params[name];
    return value === undefined ? `{${name}}` : String(value);
  });
}
```
