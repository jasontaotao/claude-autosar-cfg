// v1.27.0 — Dcm service mapper (BSWMD-driven, mirrors v1.26.0 xlsx mapper).
//
// Two-layer sheet-name mapping (the abstracted-vs-canonical seam):
//   SHEET_TO_MODULE            : user-facing xlsx sheet name → BswModuleDef key
//                                (5 entries; all map to 'Dcm' since Dcm is the
//                                only module owning Dcm service containers).
//   SHEET_TO_CONTAINER_SHORT_NAME: user-facing xlsx sheet name → canonical
//                                AUTOSAR BSWMD container shortName (DcmDsp*
//                                prefix; 4 canonical shortNames shared by
//                                the 5 user-facing sheet names — see note).
//
// Note on 5 vs 4: xlsx has 5 sheet kinds (DcmClearDTC / DcmReadDTC /
// DcmReadDataById / DcmWriteDataById / DcmRoutineControl) but canonical
// AUTOSAR uses DcmDspDid as the shared parent for both 0x22 + 0x2E.
// Mapper differentiates via per-row definitionRef emitted in PatchSteps.
//
// One row → one `add-child` + per-param `set-param` PatchSteps.
// Throws on unrecognized sheet (YAGNI guard), missing module, or missing container.

import type { PatchStep } from '../../shared/headless/ipc-contract.js';
import type { EcucInstanceRow } from '../../shared/types.js';
import { lookupContainerDef, type BswModuleDef } from '../project/bswmd.js';

export type { EcucInstanceRow };

type DcmSheetKind =
  | 'DcmClearDTC'
  | 'DcmReadDTC'
  | 'DcmReadDataById'
  | 'DcmWriteDataById'
  | 'DcmRoutineControl';

const SHEET_TO_MODULE = {
  DcmClearDTC: 'Dcm',
  DcmReadDTC: 'Dcm',
  DcmReadDataById: 'Dcm',
  DcmWriteDataById: 'Dcm',
  DcmRoutineControl: 'Dcm',
} as const;

const SHEET_TO_CONTAINER_SHORT_NAME: Readonly<Record<DcmSheetKind, string>> = {
  DcmClearDTC: 'DcmDspClearDTC',
  DcmReadDTC: 'DcmDspReadDTCInformation',
  DcmReadDataById: 'DcmDspDid',
  DcmWriteDataById: 'DcmDspDid',
  DcmRoutineControl: 'DcmDspRoutine',
};

export function xlsxDcmServicesToEcucBatch(
  rows: readonly EcucInstanceRow[],
  bswmds: ReadonlyMap<string, BswModuleDef>,
): PatchStep[] {
  const steps: PatchStep[] = [];
  for (const row of rows) {
    if (!row.shortName || row.shortName.length === 0) {
      throw new Error(`EcucInstanceRow missing shortName (sheet=${row.sheet})`);
    }
    if (!(row.sheet in SHEET_TO_MODULE)) {
      throw new Error(
        `Unrecognized sheet name: '${row.sheet}' (allowed: ${Object.keys(SHEET_TO_MODULE).join(', ')})`,
      );
    }
    const moduleShortName = SHEET_TO_MODULE[row.sheet as DcmSheetKind];
    const lookupKey = SHEET_TO_CONTAINER_SHORT_NAME[row.sheet as DcmSheetKind];
    const bswmd = bswmds.get(moduleShortName);
    if (bswmd === undefined) {
      throw new Error(
        `BSWMD map missing module '${moduleShortName}' (needed by sheet '${row.sheet}'). ` +
          `Provided modules: ${Array.from(bswmds.keys()).join(', ') || '<empty>'}`,
      );
    }
    const containerDef = lookupContainerDef(bswmd, lookupKey);
    if (containerDef === null) {
      throw new Error(
        `Container '${lookupKey}' not found in BSWMD module '${moduleShortName}'. ` +
          `Verify the BSWMD declares this canonical AUTOSAR container shortName.`,
      );
    }
    const parentPath = containerDef.path;
    const containerPath = `${parentPath}/${row.shortName}`;
    const addChildBase = {
      op: 'add-child' as const,
      parentPath,
      shortName: row.shortName,
      ...(row.definitionRef !== undefined && { definitionRef: row.definitionRef }),
    };
    steps.push(addChildBase);
    for (const [paramName, value] of Object.entries(row.params)) {
      if (value === null || value === undefined) continue;
      steps.push({
        op: 'set-param',
        containerPath,
        paramName,
        value: value as string | number | boolean,
      });
    }
  }
  return steps;
}

// v1.27.0 — Dcm config pipeline orchestrator.
//
// Loads ODX content + applies Dcm service patches (from xlsx) +
// validates cross-document ODX-Dcm references + serializes the
// merged ARXML. Reuses v1.24.0's odxToDiagnosticExtract for the
// ODX half; v1.26.0's lookupContainerDef for the BSWMD-driven
// mapper; v1.23.0's atomic-write pattern for output.

import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { parseBswmd, type BswModuleDef } from '../project/bswmd.js';
import { lookupContainerDef } from '../project/bswmd.js';
import { odxToDiagnosticExtract } from '../odxToDiagnosticExtract.js';
import type { OdxSummary } from '../../shared/types.js';
import type { EcucInstanceRow, PatchStep } from '../../shared/types.js';

export type DcmServiceKind =
  | 'DcmClearDTC'
  | 'DcmReadDTC'
  | 'DcmReadDataById'
  | 'DcmWriteDataById'
  | 'DcmRoutineControl';

export interface DcmConfigResult {
  readonly dcmConfigXml: string;
  readonly odxLinkedDcmDspCount: number;
  readonly odxLinkedRoutineCount: number;
  readonly serviceCounts: Readonly<Record<DcmServiceKind, number>>;
}

const SERVICE_KINDS: readonly DcmServiceKind[] = [
  'DcmClearDTC',
  'DcmReadDTC',
  'DcmReadDataById',
  'DcmWriteDataById',
  'DcmRoutineControl',
];

/**
 * Walk the ODX variants to collect all DID shortNames + Routine shortNames
 * referenced anywhere in the ODX. Used for the cross-document linkage check.
 */
function collectOdxDidsAndRoutines(odx: OdxSummary): {
  dcmDsps: readonly string[];
  routines: readonly string[];
} {
  // Implementer fills in once OdxSummary shape confirmed. Expected:
  // dcmDsps from variants[].dids[].shortName + dtcRef->did chains
  // routines from variants[].routines[].shortName
  const dcmDsps: string[] = [];
  const routines: string[] = [];
  for (const variant of odx.variants ?? []) {
    for (const did of variant.dids ?? []) {
      if (did.shortName) dcmDsps.push(did.shortName);
    }
    for (const routine of variant.routines ?? []) {
      if (routine.shortName) routines.push(routine.shortName);
    }
  }
  return { dcmDsps, routines };
}

/**
 * Map a xlsx row's `didRef` or `routineRef` to the corresponding ODX element
 * via the variant tree. Returns the shortName path if found, null otherwise.
 */
function resolveOdxReference(
  odx: OdxSummary,
  kind: 'didRef' | 'routineRef',
  shortName: string,
): string | null {
  // Implementer fills in once OdxSummary shape confirmed.
  for (const variant of odx.variants ?? []) {
    if (kind === 'didRef') {
      if ((variant.dids ?? []).some((d) => d.shortName === shortName)) {
        return shortName;
      }
    } else if (kind === 'routineRef') {
      if ((variant.routines ?? []).some((r) => r.shortName === shortName)) {
        return shortName;
      }
    }
  }
  return null;
}

/**
 * Validate that all xlsx rows' didRef / routineRef params resolve to ODX extract
 * shortNames. Throws with diff if not.
 */
function validateOdxLinkage(odx: OdxSummary, rows: readonly EcucInstanceRow[]): void {
  for (const row of rows) {
    if (row.sheet === 'DcmReadDataById' || row.sheet === 'DcmWriteDataById') {
      const didRef = row.params.didRef as string | undefined;
      if (didRef && resolveOdxReference(odx, 'didRef', didRef) === null) {
        const { dcmDsps } = collectOdxDidsAndRoutines(odx);
        const sample = dcmDsps.slice(0, 10).join(', ');
        const more = dcmDsps.length > 10 ? ` (and ${dcmDsps.length - 10} more)` : '';
        throw new Error(
          `ODX-Dcm linkage broken: Sheet '${row.sheet}', row '${row.shortName}': ` +
            `referenced DID '${didRef}' not found. Available DIDs from ODX: ${sample}${more}.`,
        );
      }
    }
    if (row.sheet === 'DcmRoutineControl') {
      const routineRef = row.params.routineRef as string | undefined;
      if (routineRef && resolveOdxReference(odx, 'routineRef', routineRef) === null) {
        const { routines } = collectOdxDidsAndRoutines(odx);
        throw new Error(
          `ODX-Dcm linkage broken: Sheet '${row.sheet}', row '${row.shortName}': ` +
            `referenced Routine '${routineRef}' not found. Available Routines from ODX: ${routines.join(', ') || '<none>'}.`,
        );
      }
    }
  }
}

/**
 * Convert ODX extract XML to a list of did/routine shortNames for emit-side
 * counters (extracted from the dcmContent string via regex).
 */
function countOdxElements(dcmContent: string): {
  dcmDsps: number;
  routines: number;
} {
  const dcmDsps = (dcmContent.match(/<DcmDspDid\b/g) ?? []).length;
  const routines = (dcmContent.match(/<DcmDspRoutine\b/g) ?? []).length;
  return { dcmDsps, routines };
}

export async function dcmConfigPipeline(request: {
  readonly odx: OdxSummary;
  readonly xlsxRows: readonly EcucInstanceRow[];
  readonly bswmds: ReadonlyMap<string, BswModuleDef>;
}): Promise<DcmConfigResult> {
  // 1. Validate ODX-Dcm linkage BEFORE any patches applied.
  validateOdxLinkage(request.odx, request.xlsxRows);

  // 2. Get ODX extract for Dcm (DIDs + Routines as standalone ARXML).
  const extract = odxToDiagnosticExtract({ odx: request.odx });
  const { dcmDsps: odxDcmDspCount, routines: odxRoutineCount } = countOdxElements(
    extract.dcmContent,
  );

  // 3. Apply xlsx services as PatchSteps using the Dcm BSWMD-driven mapper.
  const bswmd = request.bswmds.get('Dcm');
  if (bswmd === undefined) {
    throw new Error(
      `BSWMD map missing module 'Dcm' (needed for Dcm service configs). ` +
        `Provided modules: ${Array.from(request.bswmds.keys()).join(', ') || '<empty>'}`,
    );
  }
  // Re-route service patches through a per-row application: import the mapper
  // we create in T2 here. For now, we delegate to a separate inline strategy
  // (calling xlsxDcmServicesToEcucBatch which is in scope of T2).
  // The patches are returned; the IPC handler (T4) does the actual ARXML
  // apply+serialize. Here we just count + emit the final dcmConfigXml
  // marker; the IPC handler takes the apply-and-serialize role.
  const serviceCounts: Record<DcmServiceKind, number> = {
    DcmClearDTC: 0,
    DcmReadDTC: 0,
    DcmReadDataById: 0,
    DcmWriteDataById: 0,
    DcmRoutineControl: 0,
  };
  for (const row of request.xlsxRows) {
    if (SERVICE_KINDS.includes(row.sheet as DcmServiceKind)) {
      serviceCounts[row.sheet as DcmServiceKind] += 1;
    }
  }
  return {
    dcmConfigXml: extract.dcmContent, // T4 IPC handler applies patches + serializes final
    odxLinkedDcmDspCount: odxDcmDspCount,
    odxLinkedRoutineCount: odxRoutineCount,
    serviceCounts,
  };
}

// Reads ODX file from disk → parses via v1.22.0 parseOdxHandler →

// validation) → applies patches to a cloned ArxmlDocument →
// serializes → atomic-write with snapshot rollback.

import { readFileSync } from 'node:fs';
import { resolve as pathResolve } from 'node:path';

import { xlsxDcmServicesToEcucBatch } from '../../core/bridge/xlsxDcmServicesToEcucBatch.js';
import { dcmConfigPipeline, type DcmConfigResult } from '../../core/bridge/dcmConfigPipeline.js';
import type { EcucInstanceRow, OdxSummary } from '../../shared/types.js';
import { parseArxml } from '../../core/arxml/parser.js';

import type { PatchStep } from '../../shared/headless/ipc-contract.js';

export type IpcResult<T> =
  | { ok: true; value: T }
  | { ok: false; error: { message: string; cause?: unknown } };

/**
 * Atomic write with snapshot rollback (mirrors v1.23.0 DBC bridge pattern).
 * If a snapshot already exists, restore it before writing; if the write fails,
 * restore the snapshot.
 */
function atomicWriteWithSnapshot(outputPath: string, content: string): { restored: boolean } {
  const fs = require('node:fs') as typeof import('node:fs');
  const snapshotPath = `${outputPath}.snapshot`;
  if (fs.existsSync(outputPath)) {
    fs.renameSync(outputPath, snapshotPath);
  }
  try {
    fs.writeFileSync(outputPath, content, 'utf-8');
    if (fs.existsSync(snapshotPath)) fs.unlinkSync(snapshotPath);
    return { restored: false };
  } catch (e) {
    if (fs.existsSync(snapshotPath)) {
      fs.renameSync(snapshotPath, outputPath);
    }
    throw e;
  }
}

export async function dcmConfigHandler(args: {
  readonly odxPath: string;
  readonly xlsxRows: readonly EcucInstanceRow[];
  readonly outputPath?: string;
}): Promise<IpcResult<DcmConfigResult>> {
  try {
    // 1. Read ODX file.
    const odxXml = readFileSync(args.odxPath, 'utf-8');
    const odxParsed = parseArxml(odxXml);
    // Implementer: replace parseArxml with parseOdxHandler if shape differs.
    // For now, accept any parseable ARXML; the linkage check happens
    // inside the orchestrator.
    const odx: OdxSummary = odxParsed as unknown as OdxSummary;

    // 2. Load Dcm BSWMD from canonical demo-ecu fixture.
    //    (Real-OEM override via BSWMD override path deferred to v1.28.0.)
    const projectRoot = pathResolve(args.odxPath, '..');
    const bswmdDir = pathResolve(projectRoot, 'bswmd');
    const dcmBswmdXml = readFileSync(pathResolve(bswmdDir, 'Bsw_Dcm_Bswmd.arxml'), 'utf-8');
    const bswmds = parseDemoBswmds(new Map([['Dcm', dcmBswmdXml]]));

    // 3. Run pipeline (validates ODX-Dcm linkage, counts services).
    const pipelineResult = await dcmConfigPipeline({
      odx,
      xlsxRows: args.xlsxRows,
      bswmds,
    });

    // 4. Apply xlsx service patches to the ODX-extract-derived doc.
    //    (T3 returned the dcmConfigXml as a marker; here we serialize the
    //    final patched doc.)
    const serviceSteps = xlsxDcmServicesToEcucBatch(args.xlsxRows, bswmds);
    const extractDoc = parseArxml(pipelineResult.dcmConfigXml);
    if (!extractDoc.ok) {
      return {
        ok: false,
        error: { message: `Failed to parse ODX-extract ARXML: ${extractDoc.error.message}` },
      };
    }
    const patchedDoc = applyPatchSteps(extractDoc.value, serviceSteps);
    const finalXml = serializeArxml(patchedDoc);

    // 5. Atomic-write with snapshot rollback.
    const outputPath = args.outputPath ?? pathResolve(projectRoot, 'Dcm_Config.arxml');
    atomicWriteWithSnapshot(outputPath, finalXml);

    return {
      ok: true,
      value: {
        dcmConfigXml: finalXml,
        odxLinkedDcmDspCount: pipelineResult.odxLinkedDcmDspCount,
        odxLinkedRoutineCount: pipelineResult.odxLinkedRoutineCount,
        serviceCounts: pipelineResult.serviceCounts,
      },
    };
  } catch (e) {
    const message = e instanceof Error ? e.message : String(e);
    return { ok: false, error: { message, cause: e } };
  }
}
